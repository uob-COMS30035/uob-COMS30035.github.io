\begin{titledslide}{$k$-means optimisation}

  \begin{itemize}
  \item $r_{nk} = 1$ if datapoint $\xvec_{n}$ is assigned to cluster $k$.
  \item $\bmu_{k}$ is the mean of cluster $k$.
  \end{itemize}
  
  \begin{equation}
    \label{eq:rnk}
    r_{nk} =
    \begin{cases}
      1 & \mbox{if $k= \arg\min_{j} ||\xvec_{n} - \bmu_{j}||^{2}$} \\
      0 & \mbox{otherwise} \\
    \end{cases}
  \end{equation}

  \begin{equation}
    \label{eq:mean}
    \bmu_{k} = \frac{\sum_{n} r_{nk}\xvec_{n}}{\sum_{n} r_{nk}}
  \end{equation}

  % The algorithm minimizes:
  %   \[
  %   J = \sum_{n=1}^{N} \sum_{k=1}^{K} r_{nk} \, \lVert x_n - \mu_k \rVert^2,
  %   \]

  
\end{titledslide}

\begin{titledslide}{$k$-means objective function}

\begin{itemize}
  \item We want each point $\xvec_n$ to be close to its assigned cluster mean $\bmu_k$.
  \item Objective: minimize total squared distance to cluster centers.
\end{itemize}

\[
J = \sum_{n=1}^{N} \sum_{k=1}^{K} r_{nk} \, \lVert \xvec_n - \bmu_k \rVert^2
\]

\begin{itemize}
  \item Optimize by alternating:
    \begin{enumerate}
      \item Initialize $K$ centers.
      \item Repeat until convergence (cluster assignments stop changing).
        \begin{itemize}
            \item \textbf{E-step:} Assign each $\xvec_n$ to its nearest $\bmu_k$.
          \item \textbf{M-step:} Update each $\bmu_k$ as the mean of its assigned points. 
        \end{itemize}
          
    \end{enumerate}
  
\end{itemize}

\end{titledslide}

\begin{titledslide}{Pros and cons of $k$-means}

\textbf{Advantages}
\begin{itemize}
  \item Simple and fast: scales linearly with the number of data points $N$, clusters $K$, and dimensions $D$.
  \item Easy to implement and interpret — each cluster is summarized by its mean $\bmu_k$.
  \item Works well for roughly spherical and well-separated clusters.
  \item Often a good first choice for exploratory data analysis or initialization (e.g., for EM in GMMs).
\end{itemize}

\vskip0.5cm

\textbf{Disadvantages}
\begin{itemize}
  \item Requires pre-specifying the number of clusters $K$.
  \item Assumes clusters are isotropic (spherical) and of similar size.
  \item Sensitive to initialization — can get stuck in local minima.
  \item Sensitive to outliers and noise (means can be pulled by extreme points).
  \item Produces \emph{hard} assignments only — no uncertainty in cluster membership.
\end{itemize}

\end{titledslide}

\begin{titledslide}{Gaussian mixture distribution}


  A Gaussian mixture model (GMM) assumes that data is generated from a mixture of $K$ Gaussian components: \cite[\S 9.2]{bishop06:_patter_recog_machin_learn}

  \begin{equation}
    \label{eq:guassmix}
    p(\xvec) = \sum_{k=1}^{K} \pi_{k} \gauss(\xvec|\bmu_{k},\bcov_{k})
  \end{equation}

  We can associate the \emph{mixing coefficients} $\pi_k$ with a $K$-dimensional
  latent variable $z$ where:
  \begin{equation}
    \label{eq:zk}
    z_{k} \in \{0,1\}, \quad 
    \sum_{k} z_{k} = 1, \quad 
    p(z_{k}=1)  = \pi_{k}
  \end{equation}
  Then, the marginal likelihood can be written as:
  \begin{equation}
    \label{eq:guassmixtwo}
    p(\xvec) = \sum_{\zvec} p(\zvec) p(\xvec|\zvec) = \sum_{k=1}^{K} \pi_{k} \gauss(\xvec|\bmu_{k},\bcov_{k})
  \end{equation}

  
\end{titledslide}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%% 
\begin{titledslide}{Responsibility and sampling}

  \begin{itemize}
  \item Now we have a full joint distribution $p(\xvec,\zvec)$ we can
    define the \emph{responsibility} that component $k$ has for
    `explaining' observation \xvec
  \end{itemize}
  \begin{equation}
    \label{eq:responsibility}
    \gamma(z_{k}) = p(z_{k}=1|\xvec)
  \end{equation}

  \begin{itemize}
  \item To sample from a Gaussian mixture just use ancestral sampling:
    sample from $p(\zvec)$, and then from $p(\xvec|\zvec)$. 
  \end{itemize}
  
\end{titledslide}

\begin{titledslide}{More clustering with Gaussian mixtures}

  \begin{itemize}
  \item If we want we can put restrictions on the covariance matrices
    of the Gaussians in the mixture.
    \begin{itemize}
        \item Spherical: $\sum_k = \sigma_k^2 I$
        \item Diagonal: $\sum_k = \text{diag}(\sigma_{1}^2, \sigma_{2}^2, ..., \sigma_{K}^2)$
        \item Full: No constraint; captures correlations between features.
    \end{itemize}

  \item Let's have a look at \cite[p.729]{pml1Book}
  \end{itemize}
  
\end{titledslide}

\begin{titledslide}{(Soft) clustering by MLE of a Gaussian mixture}

  \begin{itemize}
  \item Given data \Xvec{} (and a fixed number $K$ of component Gaussians) we 
    can use MLE to estimate a particular Gaussian mixture distribution.
  \item Here's the log-likelihood
    \cite[433]{bishop06:_patter_recog_machin_learn}:
  \end{itemize}
  \begin{equation}
    \label{eq:loglik}
    \ln p(\Xvec|{\bm \pi}, \bmu, {\bm \Sigma}) = \sum_{n=1}^{N} \ln
     \left\{ \sum_{k=1}^{K} \pi_{k} \gauss(\xvec_{n}|\bmu_{k},{\bm \Sigma}_{k}) \right\}  
  \end{equation}

  \begin{itemize}
  \item Number of problems:
    \begin{enumerate}
    \item Possible singularities
    \item Symmetry/nonidentifiability
    \item No closed form for the MLE
    \end{enumerate}
  \item Queue the EM algorithm \dots
  \end{itemize}
  
\end{titledslide}



\begin{titledslide}{Comparing $k$-means and GMMs}

% \begin{center}
% \includegraphics[width=0.8\textwidth]{../figures/kmeans_vs_gmm.pdf}
% \end{center}

\begin{itemize}
  \item $k$-means: hard assignments, spherical clusters.
  \item GMM: soft assignments, elliptical clusters.
  \item GMM generalizes $k$-means when $\Sigma_k = \sigma^2 I$ and $\pi_k$ are equal.
\end{itemize}

\end{titledslide}
