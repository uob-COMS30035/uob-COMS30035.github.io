\begin{frame}[fragile]
  
\frametitle{Revisiting regression -- deterministic model}

\textbf{\textcolor{blue}{Data:}} a set of data points $D = \{ (x_1, y_1), (x_2, y_2), \cdots, (x_N, y_N) \}$ where $x_n$ is the number of rooms of house $n$ and $y_n$ the house value.
\vspace{6pt}

\textbf{\textcolor{blue}{Task:}} build a model that can predict the house value from the number of rooms
\vspace{6pt}

\uncover<2->{\textbf{\textcolor{red}{Model Type:}} parametric; assumes a polynomial relationship between house value and number of rooms}
\vspace{6pt}

\uncover<3->{\textbf{\textcolor{red}{Model Complexity:}} assume the relationship is linear $\text{house value} = a_0 + a_1 \times \text{rooms}$
\begin{equation}
y_n = a_0 + a_1 x_n
\end{equation}}
\vspace{6pt}

\uncover<4->{\textbf{\textcolor{red}{Model Parameters:}} model has two parameters $a_0$ and $a_1$ which should be estimated.

\begin{itemize}
\item $a_0$ is the y-intercept
\item $a_1$ is the slope of the line
\end{itemize}}

\end{frame}

\begin{frame}[fragile]
  
\frametitle{Revisiting linear regression -- fitting}

\begin{itemize}
\item Although we are currently assuming the model is linear, the data
  will not typically be consistent with such an assumption. So \dots
\item Find $\theta=(a_{0},a_{1})$ which minimises 
\begin{equation}
  R(a_{0},a_{1}) = \sum_{n=1}^N {(y_n - (a_{0} + a_{1}x_n))^2}
\end{equation}
\item $R$ is an abbreviation for \textcolor{blue}{residual sum of
    squares} (sometimes written $\mathrm{RSS}$).
\item Using matrix notation, then for any parameter vector $\theta$,
  which defines a linear function, we have:
\end{itemize}

\[
  \frac{\partial R}{\partial \theta} = -2\mathbf{X}^{T}(\mathbf{y}-\mathbf{X}\theta)
\]

\begin{itemize}
\item Here $X_{n,j}$ (the value at row $n$, column $j$ of
  $\mathbf{X}$) is the value of the $j$th feature in the $n$th
  datapoint (where we add a fake feature which is always 1 to handle
  the intercept), and $y_n$ is the value of the \emph{response} in the
  $n$th datapoint.
\item Setting $\frac{\partial R}{\partial \theta}$ to 0 and
  re-arranging we get:
\end{itemize}

\[
  \hat{\theta} = (\mathbf{X}^T \mathbf{X})^{-1}\mathbf{X}^T\mathbf{y}
\]


\end{frame}

  \frametitle{Computational methods for linear regression}

  \begin{itemize}
  \item Although $\hat{\theta} = (\textbf{X}^T \textbf{X})^{-1}
    \textbf{X}^T \textbf{y}$, $\hat{\theta}$ is never (well, should

  \end{itemize}

\begin{frame}[fragile]
\frametitle{Probabilistic model assumed when doing linear regression}

\begin{small}\textbf{Probabilistic models are a core part of ML}, as
  they represent uncertainty about future predictions, which is
  critical for real world applications. For simplicity, let's drop
  $a_0$ from the previous model. We add a random variable $\epsilon$
  that captures the uncertainty of predicted house prices\end{small}
\begin{equation*}
\text{house price} = a_1 \times \text{number of rooms} + \epsilon
\end{equation*}


\uncover<2->{For linear regression, we assume that $\epsilon$ is given
  by $\mathcal{N}(\mu=0, \sigma^2)$---a zero mean Gaussian
  distribution---which gives the (conditional) \textbf{\textcolor{blue}{likelihood}}
\begin{align*}
p(\mathbf{y}|\mathbf{X},a_{1},\sigma^2) &= \prod\limits_{n=1}^N{p(\text{price}_n | \text{rooms}_n, a_{1},\sigma^2)} = \prod\limits_{n=1}^N {\frac{1}{\sqrt{2 \pi} \sigma} e^{-\frac{1}{2}\frac{(\text{price}_n - a_1\text{rooms}_n)^2}{\sigma^2}}}
\end{align*}}

%\begin{equation*}
%p(\epsilon) = \frac{1}{\sqrt{2 \pi} \sigma} e^{-\frac{\epsilon^2}{2 \sigma^2}}
%\end{equation*}}

\uncover<3->{\small This distribution over $\mathbf{y}$ conditional on
  $\mathbf{X}$ is determined by the slope $a_1$ and variance $\sigma^2$ \footnote{\tiny Note that here $\mu = a_0$ which, for simplicity, we assume to be zero.}
\vspace{0pt}

\centerline{\includegraphics[scale=0.15]{../figures/probabilistic_line.png}}}

\end{frame}

\begin{titledslide}{Where's the estimate for $\sigma^2$ ?}

  \begin{itemize}
  \item When we computed the least squares fit for some data back on
    slide~\ref{ols}, no estimate for $\sigma^2$ was given.
  \item If we want $\hat{\sigma}^2$, an estimate for $\sigma^2$, it's easy
    enough to get from the observed errors:
  \[
    \hat{\sigma}^2 = \frac{1}{N-p-1}R(\hat{\theta}) = \frac{1}{N-p-1}\sum_{n=1}^{N}(y_{n}-\hat{y}_{n})^{2}
  \]
  
\item 
  $N$ is the number of datapoints, $p$ the number of estimated parameters
  and $\hat{y}_{n}$ is the predicted value for $y_{n}$.
\item We divide by $N-p-1$, rather than the more intuitive $N$, to get
  an \textbf{unbiased} estimator.
\item Idea is: if $N$ were small, but $p$ were big, then
  $R(\hat{\theta})/N$ would underestimate the uncertainty.
  $R(\hat{\theta})$ might even be 0.
\end{itemize}
\end{titledslide}

\begin{frame}[fragile]
\frametitle{Maximum Likelihood Estimation}

\begin{itemize}
\item \textbf{\textcolor{blue}{Maximum-likelihood estimation (MLE)}} is a method of estimating the parameters of a probabilistic model.
\uncover<2->{\item Assume $\boldsymbol{\theta}$  is a vector of all parameters of the probabilistic model. (e.g. $\boldsymbol{\theta} = \{a_1, \sigma^2\}$).
\item MLE produces an estimate
  $\hat{\boldsymbol{\theta}}$ of $\boldsymbol{\theta}$ by maximising
  an \textbf{\textcolor{blue}{objective function}} 
  of $\boldsymbol{\theta}$.}
\end{itemize}

\end{frame}

\begin{frame}[fragile]
\frametitle{Maximising Objective Functions}

\begin{definition}
  \begin{itemize}
  \item 
Assume $f(\theta)$ is an objective function to be optimised
(e.g. maximised), the \textcolor{blue}{$arg\,max$} value is that value
of $\theta$ which gives the maximum value of the objective function $f$:
\end{itemize}
\end{definition}
If
\vspace{8pt}
\uncover<2->{\begin{equation*}
\hat{\theta} = arg\,max_{\theta} \, f(\theta)
\end{equation*}}
then
\begin{equation*}
\forall \theta: f(\hat{\theta}) \geq f(\theta)  
\end{equation*}





\begin{itemize}
%\uncover<3->{\item \textbf{\textcolor{red}{Note:}} this is different than maximising the function (i.e. finding the maximum value [$max \, f(\theta)$])}
  \uncover<3->{
  \item Here, we are assuming that there is only one $\hat{\theta}$
    which maximises $f$. Of course, this is not always the case.
  \item  `Learning' the parameter is then equal to finding the maximum argument $arg\,max$}.
\end{itemize}

\end{frame}


\begin{frame}[fragile]
\frametitle{Maximum Likelihood Estimation - General}

\begin{itemize}
\item If $D$ is the data, then the
  \textbf{\textcolor{blue}{likelihood}} is simply $f(\theta) = P(D|\theta)$.
\begin{align*}
\theta_{MLE} &= arg\,max_\theta \, p(D|\theta)\\
 &= arg\,max_\theta \, \ln p(D|\theta)\\
 &= arg\,min_\theta \, -\ln p(D|\theta)
\end{align*}
\vspace*{-6pt}
\uncover<2->{\begin{block}{Sometimes this recipe for MLE works:}
\begin{enumerate}
\vspace{6pt}
\uncover<3->{\item Determine $\theta$, $D$ and expression for likelihood $p(D|\theta)$}
\vspace{6pt}
\uncover<4->{\item Take the natural logarithm of the likelihood}
\vspace{6pt}
\uncover<5->{\item Take the derivative of $\ln p(D|\theta)$ w.r.t. $\theta$. If $\theta$ is a multi-dimensional vector, take partial derivatives}
\vspace{6pt}
\uncover<6->{\item Set derivative(s) to 0 and solve for $\theta$}
\end{enumerate}
\end{block}}
\end{itemize}

\end{frame}

\begin{titledslide}{Least Squares and MLE for Linear Regression}

  \begin{itemize}
  \item In the case of standard linear regression one can prove that
    the parameters which minimise the squared error are also the MLE
    parameters.
  \item Note that in general the MLE recipe on the previous slide will only find
    local maxima of the likelihood (so not necessarily the MLE).
  \item But in the special case of linear regression it does find the MLE.
  \end{itemize}
  
\end{titledslide}

\begin{frame}[fragile]
\frametitle{Classification}

\begin{itemize}
\item Classification is an example of \textbf{supervised learning} \vspace{10pt}
\item Goal: Classify input data into one of $K$ classes \vspace{10pt}
\uncover<2->{\item Model: \emph{Discriminant function}:\vspace{5pt}
	\begin{itemize}
        \item A function that takes an input vector $\bm x$ and assigns it
          to class $C_k$.  For simplicity we will focus on $K=2$ and
          will first study linear functions (see Bishop for the
          general cases).
	\end{itemize}}
\end{itemize}

\end{frame}




\begin{frame}[fragile]
\frametitle{Linear discriminant function}

\begin{itemize}
\item The simplest linear discriminant (LD) is $\boxed{y(\bm{x}) = w_0 + \bm{w^Tx}}$
	\begin{itemize}
		\item \textcolor{darkgray}{where $y$ is used to predict class $C_k$, $\bm{x}$ is the input vector (feature values)
		\item $w_0$ is a scalar, which we call \emph{bias}
		\item $\bm{w}$ is our vector of parameters, which we call \emph{weights}}		
	\end{itemize}\vspace{5pt}
        \uncover<2->{\item This looks like linear regression! Except
          the next step...}  \uncover<3->{\item For $K=2$: An input
          vector $\bm x$ is assigned to class $C_1$ if $y({\bm x}) \geq 0$ and
          to class $C_2$ otherwise.}  \uncover<4->{\item A direct
          application of least-squares to choose $w_0$ and
          $\mathbf{w}$ does not give great results. See Bishop \S
          4.1.3} \uncover<5->{\item Instead we can assume the data
          from each class has a multivariate Gaussian distribution
          whose mean is class-specific (but where the covariance
          matrix for each class is the same), use MLE to find
          parameters for each of these Gaussians and finally use Bayes
          theorem to assign classes. See
          \href{https://scikit-learn.org/stable/modules/lda_qda.html}{scikit-learn
            explanation}.}
\end{itemize}
\end{frame}


\begin{frame}[fragile]
\frametitle{LD and linear separability}

\begin{itemize}
\item ``Data sets whose classes can be separated exactly by linear
  decision surfaces are said to be \emph{linearly separable}.''
  \cite[p.179]{bishop06:_patter_recog_machin_learn}.
\item We generated two sets of points using two Gaussians to
  illustrate this point, which can easily be fit by a LD.
\item A \emph{decision boundary} is the boundary that separates the
  two given classes---on one side red is more likely, on the other
  blue---which our models will try to find.
\item Note that in the data we can have a red dot where a blue is more
  likely, just by chance.
\end{itemize}

\begin{frame}[fragile]
\frametitle{Logistic regression}

\begin{itemize}
\item Assume again we have only two classes. We use a logistic function to obtain the probability of class $C_k$: $\boxed{y(\bm{x}) = \sigma(\bm{w^Tx})}$ where $\sigma$ denotes the logistic sigmoid function (s-shaped), for example:
\centerline{\includegraphics[scale=0.2]{../figures/logistic_function.png}}
\uncover<2->{
\item $\sigma(\eta) = 1/(1+\exp(-\eta))$
\item Taking a probabilistic view: 

$p(C_1|\bm{x}) = y(\bm{x})$, and $p(C_2|\bm{x}) = 1- p(C_1|\bm{x})$.}
\uncover<3->{\item NB. $\sigma$ does not represent standard deviation here!}
\end{itemize}

\end{frame}


\begin{frame}[fragile]
\frametitle{Logistic regression -- maximum likelihood estimation}

Follow MLE recipe:
\begin{enumerate}
\item Define likelihood: For a dataset $\{x_n, t_n\}$, where the
  targets $t_n \in \{0,1\}$ we have
  $p(\bm{t}|\bm{x},\bm{w}) = \mathlarger{\prod_{n=1}^N} y_n^{t_n} (1 -
  y_n)^{1-t_n}$ where $y_n = p(C_1|x_n)$.\footnote{The exponent
    selects the probability of the target class (i.e. if $t_n = 1$ we
    get $y_n$; if $t_n = 0$ we get $1-y_n$).}  \uncover<2->{ \item
    Take negative logarithm of the likelihood:
    $-\ln\:p(\bm{t}|\bm{x},\bm{w}) = - \mathlarger{\sum_{n=1}^N}\{t_n
    \ln\:y_n + (1-t_n)\ln(1-y_n)\}$} \uncover<3->{ \item Calculate the
    derivative w.r.t. the parameters $\bm{w}$:\footnote{This solution
      makes sense since we want to optimise the difference between the
      model output $y$ and the desired targets $t$.} \\
    $\frac{d\:\ln\:p(\bm{t}|\bm{x},\bm{w})}{d \bm{w}} =
    \mathlarger{\sum_{n=1}^N} (y_n - t_n)x_n$ } \uncover<4->{ \item
    Use this derivative (gradient) to iteratively improve $\bm{w}$.}
\end{enumerate}

\end{frame}

\begin{frame}[fragile]
\frametitle{MLE using Gradient Descent}
\includegraphics[scale=0.25]{../figures/grad_desc.jpeg}
\begin{itemize}
\item Start with random weight values 
\item We want to adjust each weight in $\bm w$  to minimise negative log likelihood: move downhill to the minimum
\item The derivative represents the slope: $\frac{d\:\ln\:p(\bm{t}|\bm{x},\bm{w})}{d \bm{w}} = \mathlarger{\sum_{n=1}^N} (y_n - t_n)x_n$
\item Increase or decrease $\bm w$ by a small amount in the downward
  direction
\item In the particular case of logistic regression the error function
  is \emph{convex} which means it has a unique minimum.
\item NB. The 2006 edition of the Bishop book incorrectly states that
  the error function is \emph{concave}. This was corrected in 2009.
\end{itemize}
\end{frame}


\begin{frame}[fragile]
\frametitle{Logistic regression -- maximum likelihood estimation}

More details on calculating the derivative:
\begin{enumerate}
	\item	From here $-\ln\:p(\bm{t}|\bm{x},\bm{w}) = - \mathlarger{\sum_{n=1}^N}\{t_n \ln\:y_n + (1-t_n)\ln(1-y_n)\}$
\uncover<2->{	\item We get $\mathlarger{\sum_{n=1}^N}\{ - \frac{t_n}{y_n} + \frac{(1-t_n)}{1-y_n}\} \{y_n(1-y_n)\}x_n$ \footnote{We used the chain rule and $d\: \ln(x) = 1/x$. We also used the derivative of the sigmoid $d y_n = y(1-y_n)$.}}
\uncover<3->{	\item The above simplifies to $\mathlarger{\sum_{n=1}^N}\{ - t_n (1-y_n) + (1-t_n)y_n\}x_n$}
\uncover<4->{	\item And in turn to $\mathlarger{\sum_{n=1}^N}\{ y_n - t_n \} x_n$ \footnote{You can find the full derivation \href{https://uob-coms30035.github.io/lectures/lect04/CrossEntropy.html}{here}.}}
\end{enumerate}

\end{frame}
